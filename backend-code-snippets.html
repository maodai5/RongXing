纯函数判定：同输入、同版本永远同结果

backend/app/services/triage.py

只依赖入参和规则包，不碰数据库、不调模型；结果带上规则版本，可复现、可追溯。

def triage(bundle: RulesBundle, answers: dict) -> dict:
    """红线最先；多症状取最高档。纯函数，同版本可复现。"""
    hits = []
    option_map = {o['code']: r['code'] for r in bundle.redlines['redlines'] for o in r.get('options', [])}
    selected = {s['code']: s for s in answers.get('symptoms') or []}

    for redline in bundle.redlines['redlines']:
        code = redline['code']
        explicit = any(option_map.get(o) == code for o in answers.get('redline_options') or [])
        derived = redline.get('derived')
        derived_hit = (
            bool(derived)
            and selected.get(derived['symptom'], {}).get('duration') in derived['duration']
        )
        if explicit or derived_hit:
            hits.append(code)

    grades = {}
    for code, item in selected.items():
        entry = _symptom_index(bundle)[code]
        ctx = {
            'duration': item.get('duration'),
            'frequency': item.get('frequency'),
            'spirit': answers.get('spirit'),
            'appetite': answers.get('appetite'),
            'severe': bool(item.get('severe', False)),
        }
        grades[code] = 'vet' if any(_matches(c, ctx) for c in entry.get('vet', [])) else 'observe'

    if hits:
        state = 'urgent'
    elif (
        answers.get('spirit') in bundle.symptoms['vet_triggers']['spirit']
        or answers.get('appetite') in bundle.symptoms['vet_triggers']['appetite']
    ) or 'vet' in grades.values():
        state = 'vet'
    elif grades:
        state = 'observe'
    else:
        state = 'normal'

    return {
        'state': state,
        'redline_hit': bool(hits),
        'redlines': hits,
        'symptom_grades': grades,
        'rules_version': bundle.rules_version,
    }

可靠性：幂等 + 配额

backend/app/services/idempotency.py · quota.py

幂等键挡住弱网重试；配额用数据库原子自增，和业务写同一事务，失败回滚不扣额度。

# 幂等：同键同载荷返回缓存，不同载荷 409
def find_replay(db: Session, user_id: int, key: str, request_hash: str):
    row = db.scalar(
        select(RequestIdempotency).where(
            RequestIdempotency.user_id == user_id,
            RequestIdempotency.key == key,
        )
    )
    if row is None:
        return None
    if row.request_hash != request_hash:
        raise APIError(409, 'IDEMPOTENCY_CONFLICT', '该幂等键已用于不同的请求')
    return row.response_json


# 配额：数据库原子自增，超限抛 429；失败回滚不扣额度
def consume_quota(db: Session, user_id: int, metric: str, *, now: datetime) -> int:
    limit = quota_limits()[metric]
    stmt = (
        mysql_insert(UsageCounter)
        .values(user_id=user_id, day=now.date(), metric=metric, count=1)
        .on_duplicate_key_update(count=UsageCounter.count + 1)
    )
    db.execute(stmt)
    db.flush()
    count = db.scalar(select(UsageCounter.count).where(
        UsageCounter.user_id == user_id,
        UsageCounter.day == now.date(),
        UsageCounter.metric == metric,
    ))
    if count > limit:
        raise QuotaExceeded(metric, count, limit)
    return count
